8: // Source preservation and synthesized grain can have different local texture.
9: // After falling back, a single acceptable fit is not evidence that it is safe
10: // to switch back. Keep the source until a settled rolling window remains usable
11: // for the normal 24-frame model-update interval. This state is deliberately
12: // independent of the fit history: low-confidence frames clear that history.
13: class FilmGrainSynthesisRecovery {
14: public:
15:     static constexpr int recoveryFrames = 24;
16:     static constexpr int rampFrames = 16;
17:
18:     void reset() { m_active = false; m_frames = 0; m_ramp = rampFrames; }
19:     void preserveSource() { m_active = true; m_frames = 0; m_ramp = 0; }
20:
21:     bool allow(bool modelAvailable, bool freshFit, bool settledWindow) {
22:         if (!modelAvailable) {
23:             preserveSource();
24:             return false;
25:         }
26:         if (!m_active) {
27:             if (freshFit && m_ramp < rampFrames) ++m_ramp;
28:             return true;
29:         }
30:         if (!freshFit || !settledWindow) {
31:             m_frames = 0;
32:             return false;
33:         }
34:         if (++m_frames < recoveryFrames) return false;
35:         // Past evidence cannot promise that the next frame remains usable.
36:         // Introduce synthesis gradually so an immediate new rejection does
37:         // not leave a full-strength one-frame overlay in the output.
38:         m_active = false;
39:         m_frames = 0;
40:         m_ramp = 1;
41:         return true;
42:     }
43:
44:     bool active() const { return m_active; }
45:     int frames() const { return m_frames; }
46:     double gain() const { return static_cast<double>(m_ramp) / rampFrames; }
47:
48: private:
49:     bool m_active = false;
50:     int m_frames = 0;
51:     int m_ramp = rampFrames;
52: };

54: // Apply after any existing luma residual retention. Given its coefficient r,
55: // blend an additional q of source into that base, so the combined coefficient
56: // R = r + q*(1-r) satisfies R^2 + gain^2*(1-r^2) = 1. Scaling grain without
57: // this matching source blend would temporarily remove the original texture.
58: inline double film_grain_recovery_source_blend(double gain, double retained) {
59:     gain = (std::clamp)(gain, 0.0, 1.0);
60:     retained = (std::clamp)(retained, 0.0, 1.0);
61:     if (retained == 1.0 || gain == 1.0) return 0.0;
62:     const double combined = std::sqrt(std::max(0.0, 1.0 - gain * gain * (1.0 - retained * retained)));
63:     return (std::clamp)((combined - retained) / (1.0 - retained), 0.0, 1.0);
64: }
